## 二、技术选型与架构设计

### 2.1 技术栈选择

#### 核心引擎：Phaser 3

**选择理由**：
- ✅ 成熟稳定，文档完善
- ✅ 支持WebGL加速，性能优秀
- ✅ 内置精灵动画系统
- ✅ 响应式设计支持
- ✅ 免费开源（MIT协议）
- ✅ 大量2D游戏示例

**替代方案对比**：
- **PixiJS**：渲染性能优秀，但游戏逻辑支持较弱
- **MelonJS**：专门优化移动端，但文档相对较少
- **纯Canvas API**：开发成本低，但需要自己实现大量系统

#### 前端框架：Vue 3 + Vite

**选择理由**：
- ✅ 响应式数据绑定，方便管理游戏状态
- ✅ 组件化开发，便于UI复用
- ✅ 开发体验好，热更新快速
- ✅ 与Phaser集成方便

```javascript
class PlayerSprite {
  constructor(scene) {
    this.sprite = scene.add.sprite(x, y, 'playerIdle');
    this.standby = 'playerIdle';
    this.attack = 'playerAttack';
    this.hit = 'playerHit';
    this.dead = 'playerDead';

  updateAnimation(action) {

      case 'attack':
        this.sprite.play(this.attack);
        break;
      case 'hit':

  }
}
```

### 3.2 地图系统改造

#### 3.2.1 地图渲染方案

**设计原则**：
- 逐层推进的地图系统
- 房间以网格形式显示
- 支持缩放和滚动

**实现方案**：
```javascript
class MapSystem {
  constructor(scene) {
    this.gridSize = 64; // 网格大小
    this.rooms = [];
    this.currentLayer = 1;
    this.visitedRooms = new Set();
  }

  generateLayer() {
    // 生成3个房间
    const room1 = this.generateRoom('normal');
    const room2 = this.generateRoom('elite');
    const room3 = this.generateRoom('boss');
    return [room1, room2, room3];
  }

  drawMap() {
    // 绘制房间网格
    this.rooms.forEach((room, index) => {
      const x = (index % 3) * (this.gridSize * 2);
      const y = Math.floor(index / 3) * (this.gridSize * 2);
      this.drawRoom(room, x, y);
    });
  }

  drawRoom(room, x, y) {
    const rect = this.scene.add.rectangle(x, y, this.gridSize, this.gridSize);
    rect.setStrokeStyle(2, 0xFFFFFF);

    if (room.type === 'boss') {
      rect.setFillStyle(0xFF0000); // 红色标记Boss
    } else if (room.visited) {
      rect.setFillStyle(0x00FF00); // 绿色标记已访问
    }
  }
}
```

```javascript
    this.pool = [];
    this.createFn = createFn;
    this.maxSize = maxSize;
  }

  get() {

      ? this.pool.pop()
      : this.createFn();
```

### 4.2 逻辑优化

#### 4.2.1 游戏循环优化

```javascript
class GameEngine {
  constructor(scene) {
    this.scene = scene;
    this.lastTime = 0;
    this.updateInterval = 1000 / 60; // 60fps
  }

  update(time) {
    const deltaTime = time - this.lastTime;

    if (deltaTime >= this.updateInterval) {
      this.lastTime = time;
      this.updateLogic(deltaTime);
    }

    this.scene.tweens.update();
    this.scene.physics.update();
    this.scene.events.emit('update', deltaTime);
  }

  updateLogic(deltaTime) {
    // 限制每帧计算量
    if (this.updateCounter++ % 3 === 0) {
      this.player.update(deltaTime);
      this.enemies.forEach(enemy => enemy.update(deltaTime));
    }
  }
}
```

#### 4.2.2 碰撞检测优化

**使用四叉树优化**：
```javascript
class QuadTree {
  constructor(bounds, maxObjects = 10, maxLevels = 4) {
    this.bounds = bounds;
    this.maxObjects = maxObjects;
    this.maxLevels = maxLevels;
    this.objects = [];
    this.divided = false;
  }

  insert(object) {
    // 快速判断是否在边界内
    if (!this.bounds.contains(object.bounds)) return;

    if (this.objects.length < this.maxObjects || this.level >= this.maxLevels) {
      this.objects.push(object);
    } else {
      if (!this.divided) this.divide();
      this.children.forEach(child => child.insert(object));
    }
  }

  retrieve(query) {
    let result = [];
    if (!this.bounds.contains(query.bounds)) return result;

    result.push(...this.objects);
    if (this.divided) {
      this.children.forEach(child => {
        result = result.concat(child.retrieve(query));
      });
    }

    return result;
  }
}
```

```javascript
class PerformanceMonitor {
  constructor() {
    this.isLowEnd = this.detectLowEndDevice();
    this.renderQuality = this.isLowEnd ? 'low' : 'high';
  }

  detectLowEndDevice() {
    // 检测CPU核心数
    const cores = navigator.hardwareConcurrency || 4;
    if (cores <= 2) return true;

    // 检测GPU
    const canvas = document.createElement('canvas');
    const gl = canvas.getContext('webgl') || canvas.getContext('experimental-webgl');
    if (!gl) return true;

    // 检测内存
    const memory = performance.memory;
    if (memory && memory.jsHeapSizeLimit < 1.5 * 1024 * 1024 * 1024) {
      return true;

      // 降低画质

      this.maxSprites = 100;
      this.useWebGL = false;
      this.resolution = 0.5;

      // 高画质

      this.maxSprites = 500;
      this.useWebGL = true;

}
```

### 5.3 移动端交互优化

#### 5.3.1 触摸手势支持

**基础手势**：
```javascript
class TouchGestures {
  constructor(scene) {
    this.scene.input.on('pointerdown', this.onDragStart, this);
    this.scene.input.on('pointermove', this.onDragMove, this);
    this.scene.input.on('pointerup', this.onDragEnd, this);

    this.isDragging = false;
    this.startX = 0;
    this.startY = 0;
  }

  onDragStart(pointer) {
    this.isDragging = true;
    this.startX = pointer.x;
    this.startY = pointer.y;
  }

  onDragMove(pointer) {
    if (!this.isDragging) return;

    const deltaX = pointer.x - this.startX;
    const deltaY = pointer.y - this.startY;

    // 滑动阈值检测
    if (Math.abs(deltaX) > 30 || Math.abs(deltaY) > 30) {
      this.handleSwipe(deltaX, deltaY);
      this.isDragging = false;
    }
  }

  onDragEnd() {
    this.isDragging = false;
  }

  handleSwipe(deltaX, deltaY) {
    if (Math.abs(deltaX) > Math.abs(deltaY)) {
      // 水平滑动
      if (deltaX > 0) {
        this.onSwipeRight();
      } else {
        this.onSwipeLeft();
      }
    } else {
      // 垂直滑动
      if (deltaY > 0) {
        this.onSwipeDown();
      } else {
        this.onSwipeUp();
      }
    }
  }
}
```

```javascript
    setInterval(() => this.measureFPS(), 1000);
  }

  measureFPS() {
    const now = performance.now();
    const delta = now - this.lastTime;
    this.fps = Math.round(this.frameCount * 1000 / delta);
```

#### B.2 战斗场景

```javascript
class BattleScene extends Phaser.Scene {
  constructor() {
    super({ key: 'BattleScene' });
  }

  preload() {
    this.load.setBaseURL('assets/');
    this.load.spritesheet('player', 'player.png', {
      frameWidth: 64,
      frameHeight: 64
    });
    this.load.spritesheet('enemy', 'enemy.png', {
      frameWidth: 64,
      frameHeight: 64
    });
  }

  create() {
    // 创建玩家
    this.player = this.add.sprite(200, 300, 'player', 0);
    this.playerHealth = new HealthBar(this, 200, 100, 100);

    // 创建敌人
    this.enemy = this.add.sprite(600, 300, 'enemy', 0);
    this.enemyHealth = new HealthBar(this, 600, 100, 100);

    // 添加按钮
    this.addAttackButton(200, 500);
    this.addDefenseButton(300, 500);
    this.addPotionButton(400, 500);

    // 战斗日志
    this.logText = this.add.text(400, 400, '战斗开始', {
      fontSize: '18px',
      color: '#FFFFFF',
      align: 'center'
    }).setOrigin(0.5);
  }

  addAttackButton(x, y) {
    const button = this.add.rectangle(x, y, 100, 50, 0x4CAF50);
    button.setInteractive({ useHandCursor: true });

    button.on('pointerdown', () => {
      this.playerAttack();
    });
  }

  playerAttack() {
    // 攻击动画
    this.player.play('attack');

    // 伤害计算
    const damage = Math.floor(Math.random() * 10) + 5;
    this.enemyHealth.update(this.enemyHealth.currentHp - damage, this.enemyHealth.maxHp);

    // 飘字效果
    this.showFloatingText(damage, this.enemy.x, this.enemy.y - 50);

    // 战斗日志
    this.logText.setText(`你攻击了敌人，造成${damage}点伤害`);

    // 敌人反击
    setTimeout(() => {
      this.enemyAttack();
    }, 1000);
  }

  showFloatingText(damage, x, y) {
    const text = this.add.text(x, y, `-${damage}`, {
      fontSize: '24px',
      color: '#FFFFFF'
    }).setOrigin(0.5);

    this.tweens.add({
      targets: text,
      y: y - 50,
      alpha: 0,
      duration: 1000,
      onComplete: () => text.destroy()
    });
  }

  enemyAttack() {
    const damage = Math.floor(Math.random() * 8) + 3;
    this.playerHealth.update(this.playerHealth.currentHp - damage, this.playerHealth.maxHp);

    this.showFloatingText(damage, this.player.x, this.player.y - 50);
    this.logText.setText(`敌人攻击了你，造成${damage}点伤害`);
  }
}
```
